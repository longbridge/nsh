covers [fs/promises] rename
covers [fs/promises] readlink
covers [net] default
covers [net] connect
covers [websocket] WebSocket
covers [globals] WebSocket
covers [net] Socket
covers [websocket] WebSocketConnection
# mod-accessibility
tap id:case-mod-accessibility
# no screen reader and asks for no less motion.

expect Reduce motion: false
expect Calls: not called
tap id:call
expect Calls: done
covers [nsh/accessibility] announce
covers [nsh/accessibility] setAccessibilityFocus
covers [nsh/accessibility] reduceMotion
covers [nsh/accessibility] screenReaderEnabled
# mod-android
tap id:case-mod-android

expect Exit: function
expect Calls: not called
tap id:call
expect Calls: done, remove function
covers [nsh/android] isAvailable
covers [nsh/android] BackHandler
covers [nsh/android] Toast
# mod-clipboard
tap id:case-mod-clipboard
# the headless runtime: writes go through, reads are promises that stay
# pending (and none fails).
expect Calls: not called
expect Failed: none
tap id:copy
expect Calls: 5 promises
wait 300
expect Failed: none
covers [nsh/clipboard] Clipboard
# mod-desktop
tap id:case-mod-desktop
# no title bar area, the window and menu functions do nothing, and a popup
# closes at once.
expect Desktop: false
expect Title bar: 0 0 0
expect Calls: not called
expect Popup: not shown
tap id:window
expect Calls: window done, close function, minimize function
tap id:menu
expect Calls: menu done
expect Popup: closed
tap id:toggles
expect Calls: toggles done
covers [nsh/desktop] isAvailable
covers [nsh/desktop] titleBar
covers [nsh/desktop] Window
covers [nsh/desktop] Menu

# mod-storage
tap id:case-mod-storage
# a key set twice keeps its last value, and a missing key reads null.
expect Storage empty
tap id:set
expect Storage set: a=1,b=two missing=null
tap id:remove
expect Storage removed: a=1 missing=null
tap id:clear
expect Storage cleared: none missing=null
covers [nsh/storage] Storage
