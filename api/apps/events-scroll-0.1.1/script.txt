# What a user does and sees (compat/apps.rs).
expect Scroll: none
# A drag, then a fling that glides on and comes to rest.
event id:scroll onScrollBeginDrag x=0 y=0
expect Steps: begin 0,0
scroll id:scroll 0 120
expect Scroll: 0,120 content 300x1000 size 300x200 zoom 1
event id:scroll onScrollEndDrag x=0 y=120
expect Steps: begin 0,0 | end 0,120
event id:scroll onMomentumScrollBegin x=0 y=120
expect Steps: begin 0,0 | end 0,120 | glide 0,120
scroll id:scroll 0 300.5
expect Scroll: 0,300.5 content 300x1000 size 300x200 zoom 1
event id:scroll onMomentumScrollEnd x=0 y=300.5
expect Steps: begin 0,0 | end 0,120 | glide 0,120 | stop 0,300.5
event id:scroll onScrollEnd x=0 y=300.5
expect Steps: begin 0,0 | end 0,120 | glide 0,120 | stop 0,300.5 | rest 0,300.5
covers [nsh] Scroll
covers [events] onScroll
covers [events] onScrollBeginDrag
covers [events] onScrollEndDrag
covers [events] onMomentumScrollBegin
covers [events] onMomentumScrollEnd
covers [events] onScrollEnd
