tap text:Bump home

tap text:Push

expect Page p1 at 1
expect Renders: 2
tap text:Push

expect Bumps of home: 1
# pop: one page.
tap text:Pop
expect Routes: home > p1
absent Page p2 at 2
expect Cleanups: p2
expect Renders: 3
# The same route twice is two pages; remove takes the topmost.
tap text:Push twice
expect Routes: home > p1 > twice > twice
expect Page twice at 2
expect Page twice at 3
expect Renders: 5
tap text:Remove twice
expect Routes: home > p1 > twice
expect Page twice at 2
absent Page twice at 3
expect Cleanups: p2 twice
# A route not in the stack: nothing changes.
tap text:Remove unknown

# pop(count).
tap text:Pop two
