# What a user does and sees (compat/apps.rs). What the platform reports
# (appearing, going away, Escape / back and swipes) are events here: the
# headless runtime presents nothing.
# Open by default, with the defaults.
expect Plain dialog
prop id:plain animation slide
prop id:plain presentation fullScreen
prop id:plain closeOnEscape true
prop id:plain closeOnBackdropPress true
prop id:plain closeOnSwipe true
noprop id:plain supportedOrientations
# Closed while `open` is false.
absent Synced dialog
absent Guarded dialog
tap text:Open synced
expect Synced dialog
prop id:synced animation fade
prop id:synced presentation pageSheet
listens id:synced onOpen
listens id:synced onCancel
# `style` styles the popup around the children.
frame id:synced-body 20 20 350 30
event text:"Synced dialog" onOpen
expect Log: synced opened
# Escape / back: onCancel, then onOpenChange(false, "cancel").
event text:"Synced dialog" onCancel reason=cancel
expect Log: synced opened, synced cancel, synced change false cancel
absent Synced dialog
tap text:Clear log
# Opens again when `open` goes from false to true.
tap text:Open synced
expect Synced dialog
event text:"Synced dialog" onClose
expect Log: synced closed
tap text:Close synced
absent Synced dialog
# `class` styles the popup; the iOS props.
expect Stubborn dialog
prop id:stubborn animation none
prop id:stubborn presentation formSheet
prop id:stubborn ios.supportedOrientations portrait, landscape-left
frame id:stubborn-body 12 12 366 30
# Closed by the user, it stays closed while `open` stays true.
tap text:Clear log
event text:"Stubborn dialog" onCancel reason=swipe
expect Log: stubborn change false swipe
absent Stubborn dialog
wait 100
absent Stubborn dialog
tap text:Stubborn off
absent Stubborn dialog
tap text:Stubborn on
expect Stubborn dialog
# A `closeOn*` flag set to false keeps it open: onCancel only.
tap text:Clear log
tap text:Open guarded
expect Guarded dialog
prop id:guarded closeOnEscape false
prop id:guarded closeOnBackdropPress false
prop id:guarded closeOnSwipe false
event text:"Guarded dialog" onCancel reason=cancel
expect Log: guarded cancel
event text:"Guarded dialog" onCancel reason=swipe
expect Log: guarded cancel, guarded cancel
expect Guarded dialog
tap text:Close guarded
absent Guarded dialog
# Each reason by its own flag: Escape kept, a swipe closes.
tap text:Clear log
tap text:Open swipes
expect Swipes dialog
event text:"Swipes dialog" onCancel reason=cancel
expect Swipes dialog
event text:"Swipes dialog" onCancel reason=swipe
expect Log: swipes change false swipe
absent Swipes dialog

covers [events] Dialog.onClose
covers [events] Dialog.onCancel
