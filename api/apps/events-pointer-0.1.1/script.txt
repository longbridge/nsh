# What a user does and sees (compat/apps.rs).
expect Press: none
# A finger goes down and comes up.
event id:press onPressIn
expect Press: in
event id:press onPressOut
expect Press: in out
# The mouse comes over, moves with Shift and Command held, and leaves.
listens id:hover onHoverIn
event id:hover onHoverIn
expect Hover: in
event id:hover onHoverMove locationX=5 locationY=7 shiftKey=true metaKey=true
expect Hover: move 5,7 page 25,128 meta shift
event id:hover onHoverMove locationX=6 locationY=1 altKey=true ctrlKey=true
expect Hover: move 6,1 page 26,122 alt ctrl
event id:hover onHoverOut
expect Hover: move 6,1 page 26,122 alt ctrl out
# Files dragged over, away, back and dropped.
event id:drop onFileDragEnter
expect Drop: enter
event id:drop onFileDragLeave
expect Drop: enter leave
event id:drop onFileDrop uris=file:///tmp/a.txt uris="file:///tmp/b c.png"
expect Drop: dropped 2: file:///tmp/a.txt file:///tmp/b c.png
# An assistive technology performs an action.
event id:action onAccessibilityAction actionName=increment
expect Action: increment
event id:action onAccessibilityAction actionName=activate
expect Action: activate
covers [events] onPressIn
covers [events] onPressOut
covers [events] onHoverIn
covers [events] onHoverMove
covers [events] onHoverOut
covers [events] onFileDragEnter
covers [events] onFileDragLeave
covers [events] onFileDrop
covers [events] onAccessibilityAction
