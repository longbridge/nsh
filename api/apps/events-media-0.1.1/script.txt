# What a user does and sees (compat/apps.rs).
# The platform loads the image and reports it.
expect Image:
event id:image onLoadStart
expect Image: start
event id:image onProgress loaded=512 total=2048
expect Image: start | progress 512/2048
event id:image onLoad width=640 height=480
expect Image: start | progress 512/2048 | load 640x480
event id:image onLoadEnd
expect Image: start | progress 512/2048 | load 640x480 | end
event id:broken onError message="not found"
expect Broken: error not found
# The video loads, plays, buffers, ends and fails.
event id:video onLoadStart
expect Video: start
event id:video onLoad duration=12.5 width=1920 height=1080
expect Video: start | load 12.5s 1920x1080
event id:video onProgress currentTime=3.25 duration=12.5
expect Video: start | load 12.5s 1920x1080 | at 3.25/12.5
event id:video onBuffer isBuffering=true
event id:video onBuffer isBuffering=false
expect Video: start | load 12.5s 1920x1080 | at 3.25/12.5 | buffering true | buffering false
event id:video onEnd
event id:video onError message="decode failed"
expect Video: start | load 12.5s 1920x1080 | at 3.25/12.5 | buffering true | buffering false | end | error decode failed
# The web view starts loading, navigates, loads, gets a message and fails.
event id:web onLoadStart url=https://example.com/ title="" loading=true canGoBack=false canGoForward=false
expect Web: start https://example.com/ "" loading true back false forward false
event id:web onNavigationStateChange url=https://example.com/a title="Page A" loading=false canGoBack=true canGoForward=false
event id:web onLoad url=https://example.com/a title="Page A" loading=false canGoBack=true canGoForward=false
event id:web onMessage data="hello there"
event id:web onError message="offline"
expect Web: start https://example.com/ "" loading true back false forward false | nav https://example.com/a "Page A" loading false back true forward false | load https://example.com/a "Page A" loading false back true forward false | message hello there | error offline
